Sniffy

A nose for interesting DATA in responses (HaE-style, user-extensible): emails, SSNs, Luhn-valid card PANs, phones, private IPs/MACs, ARNs, IBANs -- redacted in findings. Add a rule to CONFIG.rules and it matches immediately. Complements native credential scanning. Observe-only.

This extension declares no traffic-modifying capability. It watches requests and responses and reports findings, but does not rewrite what goes on the wire.

1// sniffy.js -- a nose for interesting data in responses (HaE-style).
2//
3// nullock:version 0.1.0
4//
5// Highlighter-and-Extractor's whole idea is a user-editable rule list that
6// tags interesting strings as they fly past -- PII, custom markers, whatever
7// this engagement cares about. Nullock's native scanner hunts *credentials*;
8// this is the complementary "flag the sensitive DATA" pass, and it's yours to
9// extend: add a rule to CONFIG.rules and it starts matching immediately.
10//
11// Observe-only: onResponse over text bodies, no permission.
12//
13// Each rule is { name, kind, severity, re, luhn? }. `re` is a RegExp; add the
14// `g` flag for multi-match. Set `luhn:true` to keep only Luhn-valid matches
15// (kills the credit-card false-positive flood). Matches are value-deduped and
16// capped per rule per response so a contacts page doesn't emit 400 findings.
17//
18// Ships with sensible PII defaults; the credential/secret rules are
19// intentionally NOT here (Nullock does those natively -- no double-reporting).

38// Luhn check (strip separators first).
39function luhnOk(s) {
40    var d = s.replace(/[ -]/g, "");
41    if (d.length < 13 || d.length > 19 || /\D/.test(d)) return false;
42    var sum = 0, alt = false;
43    for (var i = d.length - 1; i >= 0; i--) {
44        var n = d.charCodeAt(i) - 48;
45        if (alt) { n *= 2; if (n > 9) n -= 9; }
46        sum += n; alt = !alt;
47    }
48    return sum % 10 === 0;
49}

51function headerVal(headers, name) {
52    if (!headers) return "";
53    var low = name.toLowerCase();
54    for (var i = 0; i < headers.length; i++)
55        if (headers[i][0] && headers[i][0].toLowerCase() === low) return headers[i][1] || "";
56    return "";
57}
58// Redact all but the last 4 chars so we never write the full PII into a finding.
59function redact(s) {
60    s = String(s);
61    if (s.length <= 4) return "****";
62    return s.replace(/[0-9A-Za-z]/g, "*").slice(0, -4) + s.slice(-4);
63}

80        for (var r = 0; r < CONFIG.rules.length; r++) {
81            var rule = CONFIG.rules[r];
82            var re = rule.re; re.lastIndex = 0;
83            var hits = {}, count = 0, m;
84            while ((m = re.exec(body)) !== null) {
85                var val = m[0];
86                if (rule.luhn && !luhnOk(val)) { if (re.lastIndex === m.index) re.lastIndex++; continue; }
87                if (!hits[val]) { hits[val] = 1; count++; }
88                if (count >= 5000) break;                    // pathological guard
89                if (!re.global) break;
90                if (re.lastIndex === m.index) re.lastIndex++;
91            }
92            var vals = Object.keys(hits);
93            for (var i = 0; i < vals.length && i < CONFIG.perRuleCap; i++) {
94                report(rule.severity, rule.kind,
95                    rule.name + " found in response body",
96                    rule.name + ": " + redact(vals[i]), url,
97                    "sniff|" + rule.kind + "|" + url + "|" + vals[i]);
98            }
99            if (vals.length > CONFIG.perRuleCap)
100                nullock.log("sniffy: " + rule.name + " x" + vals.length + " on " + url
101                            + " (reported first " + CONFIG.perRuleCap + ")");
102        }
103    } catch (e) {
104        nullock.log("sniffy: " + (e && e.message ? e.message : String(e)));
