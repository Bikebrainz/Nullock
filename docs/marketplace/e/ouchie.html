Ouchie

Catches the app saying 'ouch': framework stack traces and debug pages (Java/.NET/Python/PHP/Ruby/Node/Go), escalating exposed interactive debuggers (Werkzeug, Whoops) to high since those are usually RCE. Observe-only.

1// ouchie.js -- catches the app saying "ouch" out loud.
2//
3// nullock:version 0.1.0
4//
5// Verbose errors are a gift to an attacker: framework stack traces leak
6// internal file paths, library versions, SQL, and sometimes source; an
7// interactive debugger page (Werkzeug console, Symfony/Whoops) can be an
8// outright RCE. Nullock natively flags SQL-error and TRACE/debug-method
9// cases, but not the broad "the server dumped a stack trace / debug page"
10// class -- so that's what this adds.
11//
12// Observe-only: onResponse over text/HTML bodies, no permission.
13//
14// Tiers:
15//   - high   : an INTERACTIVE debugger is exposed (Werkzeug debugger pin
16//              console, Symfony/Whoops, Rails web-console) -- often RCE.
17//   - medium : a framework stack trace / debug error page leaked internals
18//              (Java, .NET yellow-screen, Django DEBUG, PHP fatal + path,
19//              Ruby trace, Node trace, Spring Whitelabel with trace).
20//
21// Limits: signature-based over the 64 KiB preview. A page that merely says
22// "error" without a trace won't (and shouldn't) fire. Framework detection is
23// best-effort; the evidence snippet names what tripped it.

28function headerVal(headers, name) {
29    if (!headers) return "";
30    var low = name.toLowerCase();
31    for (var i = 0; i < headers.length; i++)
32        if (headers[i][0] && headers[i][0].toLowerCase() === low) return headers[i][1] || "";
33    return "";
34}

43// Interactive debuggers -- highest value, treat as high.
44var DEBUGGERS = [
45    { re: /Werkzeug Debugger|The debugger caught an exception|__debugger__|\?__debugger__=yes/, name: "Werkzeug interactive debugger" },
46    { re: /Whoops, looks like something went wrong|Whoops\\Run|class="Whoops|frames-container/, name: "Whoops / Symfony debug page" },
47    { re: /web-console|Rails\.application\.config\.web_console|console-plugin/, name: "Rails web-console" },
48];

50// Framework stack traces / debug pages -- medium.
51var TRACES = [
52    { re: /Traceback \(most recent call last\)/, name: "Python traceback" },
53    { re: /You(?:'|&#39;)re seeing this error because you have\s+DEBUG\s*=\s*True|django\.core\.exceptions|at django\./i, name: "Django DEBUG error page" },
54    { re: /\bat (?:java|javax|com|org)\.[\w.$]+\([\w$]+\.java:\d+\)/, name: "Java stack trace" },
55    { re: /Exception in thread \"|caused by:.*Exception/i, name: "Java exception dump" },
56    { re: /org\.springframework[\w.]*Exception|Whitelabel Error Page/, name: "Spring error" },
57    { re: /Server Error in '\/?' Application|\[[A-Za-z]*Exception:|Microsoft\.[\w.]+Exception|System\.[\w.]+Exception:/, name: ".NET / ASP.NET error" },
58    // Real .NET frames are `at <Method>(<ParamList>) in <File>:line <N>` -- the
59    // method name is ALWAYS followed by a parenthesised argument list ('()' for
60    // a parameterless method, '<Foo>d__6.MoveNext()' for async). The method-name
61    // class excludes '(', so without matching the '(...)' the required literal
62    // ' in ' is never reached and this signature was dead on every genuine
63    // trace. Match the arg list; also allow '[' ']' ',' for generic methods.
64    { re: /at [\w.<>$\[\],]+\([^)]*\) in [A-Za-z]:\\[^\n]+:line \d+/, name: ".NET stack trace" },
65    { re: /(?:Fatal error|Parse error|Warning|Notice)\s*:\s*.+ in \/?[\w./-]+\.php on line \d+/i, name: "PHP error with path" },
66    { re: /#\d+ \/[\w./-]+\(\d+\):|Stack trace:\s*#0/, name: "PHP stack trace" },
67    { re: /(?:app\/controllers\/|\/gems\/|activerecord|actionpack)[\w./-]*\.rb:\d+/, name: "Ruby / Rails trace" },
68    { re: /\bat (?:Object\.|Module\.|Function\.)?[\w.<>$]*\s*\(?[\w./-]*node_modules[\w./-]+:\d+:\d+\)?/, name: "Node.js stack trace" },
69    { re: /goroutine \d+ \[[^\]]+\]:|panic: .+\n\s+\/[\w./-]+\.go:\d+/, name: "Go panic trace" },
70];

72nullock.onResponse(function(entry) {
73    try {
74        var ct = headerVal(entry.headers, "Content-Type").toLowerCase();
75        // Only meaningful for text-ish bodies.
76        if (ct && !/text|html|json|xml|plain/.test(ct)) return entry;
77        var body = entry.bodyText || entry.bodyPreview || "";
78        if (!body || body.length < 24) return entry;
79        var url = entry.url || "";
80        var i;
81​
82        for (i = 0; i < DEBUGGERS.length; i++) {
83            if (DEBUGGERS[i].re.test(body)) {
84                report("high", "debug-console-exposed",
85                    "Interactive debugger exposed (" + DEBUGGERS[i].name + ") -- typically leads to RCE",
86                    DEBUGGERS[i].name + " markers in response body", url, "dbg|" + url + "|" + i);
87                return entry; // one is plenty
88            }
89        }
90        for (i = 0; i < TRACES.length; i++) {
91            if (TRACES[i].re.test(body)) {
92                report("medium", "verbose-error-disclosure",
93                    "Verbose error / stack trace disclosed (" + TRACES[i].name + ") -- leaks paths, versions, internals",
94                    TRACES[i].name + " matched in response body", url, "trace|" + url + "|" + i);
95                return entry;
96            }
97        }
98    } catch (e) {
99        nullock.log("ouchie: " + (e && e.message ? e.message : String(e)));
100    }
101    return entry;
102});
