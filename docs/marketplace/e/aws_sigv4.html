99function strToBytes(s) {
100    // UTF-8 encode.
101    var out = [];
102    for (var i = 0; i < s.length; i++) {
103        var c = s.charCodeAt(i);
104        if (c < 0x80) out.push(c);
105        else if (c < 0x800) { out.push(0xc0|(c>>6)); out.push(0x80|(c&0x3f)); }

138function detectService(host) {
139    // s3.amazonaws.com, s3-us-east-1.amazonaws.com -> s3
140    // bucket.s3.amazonaws.com -> s3
141    // dynamodb.us-east-1.amazonaws.com -> dynamodb
142    // lambda.us-east-1.amazonaws.com -> lambda
143    var stripped = host.toLowerCase();
144    var parts = stripped.split(".");
145    for (var i = 0; i < parts.length; i++) {
146        if (parts[i].indexOf("amazonaws") >= 0) {
147            // Walk back: typical shape "<service>.<region>.amazonaws.com"
148            // or "<service>-<region>.amazonaws.com" or just "<service>.amazonaws.com"
149            if (i >= 2) return parts[i-2].split("-")[0];
150            if (i >= 1) return parts[i-1].split("-")[0];
151        }
152    }
153    return "execute-api";  // safe fallback (api-gateway)
154}
