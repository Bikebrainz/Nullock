Jelly

Spots JSONP endpoints and cross-site JSON-hijacking shapes -- reflected callback wrappers (cross-site readable), arbitrary-callback reflection (XSS), and top-level JSON arrays. The callback angle native CORS checks don't cover. Observe-only.

1// jelly.js -- wobbly JSON: JSONP endpoints & cross-site JSON hijacking.
2//
3// nullock:version 0.1.0
4//
5// A JSONP endpoint wraps its data in an attacker-named callback, so any site
6// can <script src> it and read the response cross-origin -- and if the
7// callback name is reflected raw, that's reflected XSS on top. Top-level JSON
8// arrays are the classic array-hijack shape. Nullock covers CORS natively but
9// nothing looks at the JSONP/callback angle, so this fills it.
10//
11// Observe-only: onResponse, reads the request URL's query + the response body
12// and content-type. No permission.
13//
14// What it reports:
15//   - jsonp-callback-xss (high): the callback value is reflected verbatim AND
16//     carries HTML/JS metacharacters -> reflected XSS via the callback param.
17//   - jsonp-endpoint (medium): the response is a reflected callback wrapper
18//     around data -> cross-site readable; confirm it returns anything sensitive.
19//   - json-array-hijack (low): a top-level JSON array served 200 -> legacy
20//     array-constructor hijack shape; low on modern browsers, still worth noting.
21//
22// Limits: passive and single-message -- it flags the shape, it can't prove the
23// body is sensitive or that a victim is authenticated. Leads, not proof.

28function headerVal(headers, name) {
29    if (!headers) return "";
30    var low = name.toLowerCase();
31    for (var i = 0; i < headers.length; i++)
32        if (headers[i][0] && headers[i][0].toLowerCase() === low) return headers[i][1] || "";
33    return "";
34}
35function clip(s, n) { s = String(s || ""); return s.length > n ? s.substring(0, n) + "..." : s; }
36​
37function queryParam(url, names) {
38    var qm = url.indexOf("?"); if (qm < 0) return null;
39    var qs = url.substring(qm + 1); var h = qs.indexOf("#"); if (h >= 0) qs = qs.substring(0, h);
40    var parts = qs.split("&");
41    for (var i = 0; i < parts.length; i++) {
42        var eq = parts[i].indexOf("=");
43        var k = (eq >= 0 ? parts[i].substring(0, eq) : parts[i]).toLowerCase();
44        if (names.indexOf(k) >= 0) {
45            var v = eq >= 0 ? parts[i].substring(eq + 1) : "";
46            try { v = decodeURIComponent(v.replace(/\+/g, " ")); } catch (e) {}
47            return v;
48        }
49    }
50    return null;
51}

60var CALLBACK_PARAMS = ["callback", "jsonp", "jsonpcallback", "cb", "json.callback", "callbackname", "jsoncallback"];

62nullock.onResponse(function(entry) {
63    try {
64        if (entry.status && (entry.status < 200 || entry.status >= 300)) return entry;
65        var url = entry.url || "";
66        var ct = headerVal(entry.headers, "Content-Type").toLowerCase();
67        var body = entry.bodyText || entry.bodyPreview || "";
68        if (!body) return entry;
69​
70        var head = body.replace(/^﻿/, "").replace(/^\s*\/\*[\s\S]*?\*\/\s*/, "").replace(/^[\s;]+/, "");
71        var isJsCt = /javascript|ecmascript/.test(ct);
72​
73        // JSONP: leading `name(` wrapper.
74        var wrap = head.match(/^([A-Za-z_$][\w$.\[\]'"]{0,120})\s*\(/);
75        var cbVal = queryParam(url, CALLBACK_PARAMS);
76        if (wrap && (isJsCt || cbVal)) {
77            var fn = wrap[1];
78            var reflected = cbVal && head.indexOf(cbVal) === 0;   // callback value drives the wrapper name
79            if (reflected && /[<>"'();=\s]/.test(cbVal)) {
80                report("high", "jsonp-callback-xss",
81                    "JSONP callback reflected verbatim with JS/HTML metacharacters -- reflected XSS via the callback parameter",
82                    "callback=" + clip(cbVal, 80) + "  ->  body starts: " + clip(head, 80), url,
83                    "cbxss|" + url.split("?")[0]);
84            } else if (reflected || (isJsCt && /^[\w$.]+\($/.test(fn + "("))) {
85                report("medium", "jsonp-endpoint",
86                    "JSONP endpoint (callback-wrapped response) -- cross-site readable via <script src>",
87                    "wrapper: " + clip(fn, 60) + "(...)" + (cbVal ? ", callback param reflected" : ""), url,
88                    "jsonp|" + url.split("?")[0]);
89            }
90            return entry;
91        }
92​
93        // Top-level JSON array (array-hijack shape) -- only when it's really JSON.
94        if (/json/.test(ct) && /^\s*\[/.test(body) && /[\{"]/.test(body)) {
95            report("low", "json-array-hijack",
96                "Top-level JSON array response -- legacy cross-site array-hijack shape; prefer an object or an anti-hijack prefix",
97                "body starts: " + clip(body.replace(/\s+/g, " "), 60), url,
98                "arr|" + url.split("?")[0]);
99        }
100    } catch (e) {
101        nullock.log("jelly: " + (e && e.message ? e.message : String(e)));
102    }
103    return entry;
104});
