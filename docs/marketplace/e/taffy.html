Taffy

Inline <@tag>..<@/tag> transforms resolved on outgoing requests -- base64/url/hex/html encode+decode, sha256/sha1, upper/lower/reverse, timestamp, random, uuid, repeat. Keep payloads readable and let the tag encode on the wire; tags nest innermost-first. (Nullock's take on Hackvertor.)

1// taffy.js -- inline tag-based transforms on outgoing requests.
2//
3// nullock:permissions modify-requests
4//
5// A proxy-side port of Gareth Heyes' Hackvertor (one of the most-installed
6// BApps). Hackvertor lets you drop <@tag>...<@/tag> markup into a request
7// and have Burp substitute the encoded/hashed/generated value on the way
8// out -- so you can keep a payload readable in Repeater and let the tool do
9// the base64 / URL-encoding / hashing / timestamping for you.
10//
11// Nullock has no per-request editor tab, so we do the same thing at the
12// proxy: every outgoing request's path, header VALUES and body are scanned
13// for <@...@> tags and rewritten before they hit the wire. Type a tag
14// anywhere you can type a request (Repeater, a crafted client, a fuzzer
15// template) and it resolves.
16//
17// Supported tags (a faithful subset of Hackvertor's set):
18//
19//   Encoders (paired):   <@base64>x<@/base64>  <@b64d>x<@/b64d>
20//                        <@urlenc>x<@/urlenc>  <@urldec>x<@/urldec>
21//                        <@hex>x<@/hex>        <@unhex>x<@/unhex>
22//                        <@htmlent>x<@/htmlent>
23//   Case / order:        <@upper> <@lower> <@reverse>
24//   Hashes (paired):     <@sha256>x<@/sha256>  <@sha1>x<@/sha1>
25//   Generators (self-closing):
26//                        <@timestamp/>    unix seconds, now
27//                        <@timestamp_ms/> unix millis, now
28//                        <@random(N)/>    N hex chars (default 8)
29//                        <@uuid/>         random v4 UUID
30//   Repeat (paired):     <@repeat(3)>ab<@/repeat>  -> ababab
31//
32// Tags nest: <@base64><@sha256>hunter2<@/sha256><@/base64> hashes then
33// base64s the digest. Innermost tags resolve first.
34//
35// Honest limits: QJSEngine ships no crypto, so SHA-1/256 are inlined here
36// (correct, but not constant-time -- fine for test payloads, don't build a
37// production signer on it). Only the tags above are implemented; unknown
38// tags are left untouched so they're visible rather than silently dropped.

44function strToBytes(s) {
45    var out = [];
46    for (var i = 0; i < s.length; i++) {
47        var c = s.charCodeAt(i);
48        if (c < 0x80) out.push(c);
49        else if (c < 0x800) { out.push(0xc0 | (c >> 6)); out.push(0x80 | (c & 0x3f)); }
50        else if ((c & 0xfc00) === 0xd800) {
51            var c2 = s.charCodeAt(++i);
52            var cp = 0x10000 + (((c & 0x3ff) << 10) | (c2 & 0x3ff));
53            out.push(0xf0 | (cp >> 18)); out.push(0x80 | ((cp >> 12) & 0x3f));
54            out.push(0x80 | ((cp >> 6) & 0x3f)); out.push(0x80 | (cp & 0x3f));
55        } else {
56            out.push(0xe0 | (c >> 12)); out.push(0x80 | ((c >> 6) & 0x3f));
57            out.push(0x80 | (c & 0x3f));
58        }
59    }
60    return out;

62function bytesToStr(b) {
63    var s = "", i = 0;
64    while (i < b.length) {
65        var c = b[i++];
66        if (c < 0x80) s += String.fromCharCode(c);
67        else if (c >= 0xc0 && c < 0xe0) s += String.fromCharCode(((c & 0x1f) << 6) | (b[i++] & 0x3f));
68        else if (c >= 0xe0 && c < 0xf0)
69            s += String.fromCharCode(((c & 0x0f) << 12) | ((b[i++] & 0x3f) << 6) | (b[i++] & 0x3f));
70        else {

82function b64encodeBytes(bytes) {
83    var out = "";
84    for (var i = 0; i < bytes.length; i += 3) {
85        var b0 = bytes[i], b1 = i + 1 < bytes.length ? bytes[i + 1] : 0,
86            b2 = i + 2 < bytes.length ? bytes[i + 2] : 0;
87        out += B64[b0 >> 2];
88        out += B64[((b0 & 3) << 4) | (b1 >> 4)];
89        out += i + 1 < bytes.length ? B64[((b1 & 15) << 2) | (b2 >> 6)] : "=";
90        out += i + 2 < bytes.length ? B64[b2 & 63] : "=";
91    }
92    return out;
93}
94function b64decodeToBytes(str) {
95    // The `=` padding is stripped by the filter above, so a final group can
96    // legitimately hold 2 or 3 chars; absent positions decode as 0 and are
97    // NOT emitted (indexing past the string would give indexOf() -1 and
98    // corrupt the last byte).
99    var clean = String(str).replace(/[^A-Za-z0-9+/]/g, ""), out = [];
100    for (var i = 0; i < clean.length; i += 4) {
101        var e0 = B64.indexOf(clean[i]), e1 = B64.indexOf(clean[i + 1]);
102        var has2 = i + 2 < clean.length, has3 = i + 3 < clean.length;
103        var e2 = has2 ? B64.indexOf(clean[i + 2]) : 0;
104        var e3 = has3 ? B64.indexOf(clean[i + 3]) : 0;
105        var n = (e0 << 18) | (e1 << 12) | ((e2 < 0 ? 0 : e2) << 6) | (e3 < 0 ? 0 : e3);
106        out.push((n >> 16) & 0xff);
107        if (has2) out.push((n >> 8) & 0xff);
108        if (has3) out.push(n & 0xff);
109    }
110    return out;
111}

179    function rol(x, n) { return ((x << n) | (x >>> (32 - n))) >>> 0; }
180    for (var blk = 0; blk < m.length; blk += 64) {

211var TAGS = {
212    base64:  function(x){ return b64encodeBytes(strToBytes(x)); },
213    b64d:    function(x){ return bytesToStr(b64decodeToBytes(x)); },
214    urlenc:  function(x){ return encodeURIComponent(x); },
215    urldec:  function(x){ try { return decodeURIComponent(x); } catch (e) { return x; } },
216    hex:     function(x){ return bytesToHex(strToBytes(x)); },
217    unhex:   function(x){ return bytesToStr(hexToBytes(x)); },
218    htmlent: function(x){ return htmlEncode(x); },
219    upper:   function(x){ return x.toUpperCase(); },
220    lower:   function(x){ return x.toLowerCase(); },
221    reverse: function(x){ return x.split("").reverse().join(""); },
222    sha256:  function(x){ return bytesToHex(sha256bytes(strToBytes(x))); },
223    sha1:    function(x){ return bytesToHex(sha1bytes(strToBytes(x))); },
224    repeat:  function(x, arg){ var n = Math.max(0, parseInt(arg||"1",10)||0); var o=""; while(n-->0) o+=x; return o; },

238function applyTags(text) {
239    if (!text || text.indexOf("<@") < 0) return text;
240    var out = text, passes = 0;
241    while (passes++ < 200) {
242        var changed = false, m;
243        if ((m = PAIRED.exec(out)) !== null) {
244            var fn = TAGS[m[1].toLowerCase()];
245            var rep = fn ? fn(m[3], m[2]) : m[0];
246            out = out.substring(0, m.index) + rep + out.substring(m.index + m[0].length);
247            changed = true;
248        } else if ((m = SELF.exec(out)) !== null) {
249            var sfn = TAGS[m[1].toLowerCase()];
250            var srep = sfn ? sfn("", m[2]) : m[0];
251            out = out.substring(0, m.index) + srep + out.substring(m.index + m[0].length);
252            changed = true;
253        }
254        if (!changed) break;
255    }
256    return out;
257}

259nullock.onRequest(function(req) {
260    try {
261        // Path (incl. query string), every header value, and the body.
262        if (req.path && req.path.indexOf("<@") >= 0) req.path = applyTags(req.path);
263        for (var i = 0; i < req.headers.length; i++) {
264            if (req.headers[i][1] && req.headers[i][1].indexOf("<@") >= 0)
265                req.headers[i][1] = applyTags(req.headers[i][1]);
266        }
267        if (req.bodyText && req.bodyText.indexOf("<@") >= 0)
268            req.bodyText = applyTags(req.bodyText);
269        return req;
270    } catch (e) {
271        nullock.log("taffy: " + (e && e.message ? e.message : String(e)));
272        return req;
273    }
