Waltz

Watches the OAuth 2.0 / OIDC dance and flags the classic missteps -- implicit flow, missing state, missing PKCE, tokens leaking into request URLs or 3xx Location headers, cleartext redirect_uri. The flow around the token that native JWT checks don't see. Observe-only.

1// waltz.js -- passive OAuth 2.0 / OpenID Connect misconfig auditor.
2//
3// The proxy-side answer to EsPReSSO / OAuthScan: watch the OAuth dance go
4// by and flag the classic mistakes without sending a single extra request.
5// Nullock's native scanner already dissects JWTs; this covers the *flow*
6// around them -- how the authorization request is built and how tokens come
7// back -- which nothing native looks at.
8//
9// Observe-only: it registers onResponse and reads the request URL (query
10// included), the response status + Location header, and the body preview.
11// It never mutates and needs no permission.
12//
13// What it catches:
14//   - Implicit flow: response_type=token / id_token (tokens delivered in the
15//     URL, cacheable, logged, in Referer -- deprecated by OAuth 2.1).
16//   - Missing `state`: no CSRF defence on the authorization request.
17//   - Missing PKCE: an authorization-code request with no code_challenge
18//     (mandatory for public clients under OAuth 2.1).
19//   - Tokens in the URL: access_token / id_token / code landing in a query
20//     string or a redirect Location (leaks via history, Referer, logs).
21//   - Cleartext redirect_uri: an http:// callback for a bearer-token flow.
22//
23// Limits: purely passive and single-message -- it can't confirm redirect_uri
24// allow-listing or replay `code`. Treat findings as leads, per the BApp it
25// mirrors. `client_id` is truncated in evidence; no secrets are logged.

32function parseQuery(qs) {
33    var out = {};
34    if (!qs) return out;
35    var parts = qs.split("&");
36    for (var i = 0; i < parts.length; i++) {
37        var eq = parts[i].indexOf("=");
38        var k = eq >= 0 ? parts[i].substring(0, eq) : parts[i];
39        var v = eq >= 0 ? parts[i].substring(eq + 1) : "";
40        try { k = decodeURIComponent(k); } catch (e) {}
41        try { v = decodeURIComponent(v.replace(/\+/g, " ")); } catch (e2) {}
42        if (k) out[k.toLowerCase()] = v;
43    }
44    return out;
45}

47function splitUrl(url) {
48    var q = "", frag = "", u = url || "";
49    var h = u.indexOf("#");
50    if (h >= 0) { frag = u.substring(h + 1); u = u.substring(0, h); }
51    var qm = u.indexOf("?");
52    if (qm >= 0) { q = u.substring(qm + 1); u = u.substring(0, qm); }
53    return { base: u, query: q, fragment: frag };
54}

56function headerVal(headers, name) {
57    if (!headers) return "";
58    var low = name.toLowerCase();
59    for (var i = 0; i < headers.length; i++)
60        if (headers[i][0] && headers[i][0].toLowerCase() === low) return headers[i][1] || "";
61    return "";
62}

74// Does this URL/fragment carry a token or auth code?
75function tokenLeak(part) {
76    if (!part) return "";
77    if (/(?:^|[&?#])access_token=/.test(part))  return "access_token";
78    if (/(?:^|[&?#])id_token=/.test(part))      return "id_token";
79    if (/(?:^|[&?#])code=[^&]{8,}/.test(part))  return "authorization code";
80    return "";
81}

83function auditAuthorize(url, q) {
84    // Only treat it as an OAuth authorize request if it smells like one.
85    var looksOAuth = ("client_id" in q) && (("redirect_uri" in q) || ("response_type" in q) || ("scope" in q));
86    if (!looksOAuth) return;
87​
88    var rt = (q["response_type"] || "").toLowerCase();
89    var cid = clip(q["client_id"], 24);
90    var ruri = q["redirect_uri"] || "";
91​
92    if (/\btoken\b/.test(rt) || rt.indexOf("id_token") >= 0) {
93        report("medium", "oauth-implicit-flow",
94            "OAuth implicit flow in use (response_type=" + rt + "): tokens are returned in the URL",
95            "authorize request, client_id=" + cid + ", response_type=" + rt, url,
96            "impl|" + splitUrl(url).base + "|" + rt);
97    }
98    if (!("state" in q)) {
99        report("medium", "oauth-missing-state",
100            "OAuth authorization request has no `state` parameter (no CSRF binding)",
101            "authorize request, client_id=" + cid, url,
102            "state|" + splitUrl(url).base);
103    }
104    if (rt.indexOf("code") >= 0 && !("code_challenge" in q)) {
105        report("medium", "oauth-missing-pkce",
106            "Authorization-code request without PKCE (no code_challenge)",
107            "authorize request, client_id=" + cid + ", response_type=" + rt, url,
108            "pkce|" + splitUrl(url).base + "|" + cid);
109    }
110    if (/^http:\/\//i.test(ruri) && (/\btoken\b/.test(rt) || rt.indexOf("code") >= 0)) {
111        report("medium", "oauth-cleartext-redirect-uri",
112            "OAuth redirect_uri is cleartext http:// -- credentials/code exposed on the callback",
113            "redirect_uri=" + clip(ruri, 80), url,
114            "ruri|" + splitUrl(url).base + "|" + clip(ruri, 40));
115    }
116}

118nullock.onResponse(function(entry) {
119    try {
120        var url = entry.url || "";
121        var parts = splitUrl(url);
122        var q = parseQuery(parts.query);
123​
124        // 1. Audit an authorization request as it goes out.
125        if (AUTHZ_PATH.test(parts.base) || ("response_type" in q) || ("client_id" in q))
126            auditAuthorize(url, q);
127​
128        // 2. Token/code leaking into THIS request's own URL (query or fragment
129        //    server-side can't see the fragment, but if we can it's exposure).
130        var inUrl = tokenLeak(parts.query) || tokenLeak(parts.fragment);
131        if (inUrl) {
132            report("high", "oauth-token-in-url",
133                "OAuth " + inUrl + " present in a request URL (leaks via history, Referer and logs)",
134                "location: request URL", url, "urltok|" + parts.base + "|" + inUrl);
135        }
136​
137        // 3. Token/code leaking through a redirect Location.
138        if (entry.status >= 300 && entry.status < 400) {
139            var loc = headerVal(entry.headers, "Location");
140            var inLoc = tokenLeak(loc);
141            if (inLoc) {
142                report("high", "oauth-token-in-redirect",
143                    "OAuth " + inLoc + " returned in a 3xx Location header (URL-borne token leakage)",
144                    "Location: " + clip(loc, 100), url,
145                    "loctok|" + parts.base + "|" + inLoc);
146            }
147        }
148    } catch (e) {
149        nullock.log("waltz: " + (e && e.message ? e.message : String(e)));
150    }
151    return entry;
152});
