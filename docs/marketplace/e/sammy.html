Sammy

Pulls the base64 SAMLResponse out of an IdP auto-POST form and audits it -- unsigned assertions, XML comment-injection surface (CVE-2017-11427 class), over-long validity, missing InResponseTo/Destination, cleartext ACS. (Passive half of SAML Raider.) Observe-only.

1// sammy.js -- passive SAML response/assertion auditor.
2//
3// nullock:version 0.1.0
4//
5// A proxy-side take on SAML Raider (the SAML BApp everyone reaches for).
6// SAML Raider is an active editor for forging/replaying assertions; Nullock
7// can't hook the browser to intercept the POST, but it doesn't need to: the
8// IdP hands the browser an auto-submitting HTML form carrying the base64
9// SAMLResponse, and that form flows through us in a *response* body. So we
10// grab it there, base64-decode it (POST binding is plain base64 -- no
11// DEFLATE, unlike the redirect binding), and audit the XML.
12//
13// Observe-only: onResponse, no mutation, no permission.
14//
15// What it catches (the high-value SAML mistakes):
16//   - Unsigned assertion / response: no <Signature> => the SP is trusting an
17//     attacker-forgeable document. The #1 SAML finding.
18//   - XML comment-injection surface: a comment inside NameID/attributes is
19//     the CVE-2017-11427 / "SAML is insecure by design" truncation class.
20//   - Over-long validity window: NotOnOrAfter far out, or NotBefore absent,
21//     widening the replay window.
22//   - Missing Destination / InResponseTo: no audience/flow binding, easing
23//     token reuse and cross-SP replay.
24//   - Cleartext ACS / Destination (http://): assertion delivered in the clear.
25//
26// Limits: POST-binding only (redirect-binding SAML is DEFLATE-compressed and
27// QJSEngine has no inflate). Signature PRESENCE is checked, not signature
28// VALIDITY or what it covers -- a present-but-scoped-wrong signature still
29// warrants manual review. Only the first ~64 KiB of the body is visible.

34// ---- base64 decode (no atob in QJSEngine) -> UTF-8 string -------------
35var B64 = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
36function b64ToStr(str) {
37    var clean = String(str).replace(/[^A-Za-z0-9+/]/g, ""), bytes = [];
38    for (var i = 0; i < clean.length; i += 4) {
39        var has2 = i + 2 < clean.length, has3 = i + 3 < clean.length;
40        var e0 = B64.indexOf(clean[i]), e1 = B64.indexOf(clean[i + 1]);
41        var e2 = has2 ? B64.indexOf(clean[i + 2]) : 0, e3 = has3 ? B64.indexOf(clean[i + 3]) : 0;
42        var n = (e0 << 18) | (e1 << 12) | ((e2 < 0 ? 0 : e2) << 6) | (e3 < 0 ? 0 : e3);
43        bytes.push((n >> 16) & 0xff);
44        if (has2) bytes.push((n >> 8) & 0xff);
45        if (has3) bytes.push(n & 0xff);
46    }
47    // UTF-8 decode
48    var s = "", j = 0;
49    while (j < bytes.length) {
50        var c = bytes[j++];
51        if (c < 0x80) s += String.fromCharCode(c);
52        else if (c >= 0xc0 && c < 0xe0) s += String.fromCharCode(((c & 0x1f) << 6) | (bytes[j++] & 0x3f));
53        else if (c >= 0xe0 && c < 0xf0)
54            s += String.fromCharCode(((c & 0x0f) << 12) | ((bytes[j++] & 0x3f) << 6) | (bytes[j++] & 0x3f));
55        else j += 3; // skip astral; SAML XML is ~ASCII
56    }
57    return s;
58}

69// Pull the base64 SAML blob out of an HTML auto-post form or a urlencoded body.
70function extractSaml(body) {
71    var out = [];
72    // <input ... name="SAMLResponse" ... value="BASE64" ...> (attr order varies)
73    var re = /name\s*=\s*["']?(SAMLResponse|SAMLRequest)["']?[^>]*?value\s*=\s*["']([^"']+)["']/gi, m;
74    while ((m = re.exec(body)) !== null) out.push({ kind: m[1], b64: m[2] });
75    // value-before-name ordering
76    var re2 = /value\s*=\s*["']([A-Za-z0-9+/=\s]{40,})["'][^>]*?name\s*=\s*["']?(SAMLResponse|SAMLRequest)/gi;
77    while ((m = re2.exec(body)) !== null) out.push({ kind: m[2], b64: m[1] });
78    // form-urlencoded (rare in a response, but cheap to check)
79    var re3 = /(?:^|[&?])(SAMLResponse|SAMLRequest)=([A-Za-z0-9%+/=]{40,})/gi;
80    while ((m = re3.exec(body)) !== null) {
81        var v = m[2]; try { v = decodeURIComponent(v.replace(/\+/g, " ")); } catch (e) {}
82        out.push({ kind: m[1], b64: v });
83    }
84    return out;
85}

87function auditXml(xml, kind, url, tokenTag) {
88    var isResponse = kind === "SAMLResponse";
89​
90    // 1. Signature presence. An unsigned Response/Assertion is the big one.
91    if (!/<(?:\w+:)?Signature\b/.test(xml)) {
92        report(isResponse ? "high" : "medium", "saml-unsigned",
93            "SAML " + kind + " has no <Signature> -- assertion is forgeable/tamperable",
94            "decoded " + kind + " (" + xml.length + " bytes), no Signature element", url,
95            "sig|" + tokenTag);
96    }
97​
98    // 2. XML comment inside identity-bearing nodes => comment-truncation surface.
99    if (/<!--/.test(xml)) {
100        var nearId = /<(?:\w+:)?(?:NameID|Attribute|AttributeValue|Subject)\b[^>]*>[^<]*<!--/.test(xml)
101                  || /<!--[\s\S]{0,80}(?:NameID|AttributeValue)/.test(xml);
102        report(nearId ? "high" : "medium", "saml-comment-injection-surface",
103            "XML comment present in the SAML document (CVE-2017-11427-class comment-truncation surface)"
104            + (nearId ? " near an identity node" : ""),
105            "comment found in decoded " + kind, url, "cmt|" + tokenTag);
106    }
107​
108    // 3. Validity window.
109    var naMatch = xml.match(/NotOnOrAfter\s*=\s*["']([^"']+)["']/);
110    var nbPresent = /NotBefore\s*=/.test(xml);
111    if (naMatch) {
112        var na = Date.parse(naMatch[1]);
113        if (!isNaN(na)) {
114            var hours = (na - Date.now()) / 3600000;
115            if (hours > 24)
116                report("low", "saml-long-validity",
117                    "SAML assertion valid for ~" + Math.round(hours) + "h (wide replay window)",
118                    "NotOnOrAfter=" + naMatch[1], url, "val|" + tokenTag);
119        }
120    }
121    if (isResponse && !nbPresent)
122        report("low", "saml-no-notbefore",
123            "SAML assertion has no NotBefore constraint",
124            "no NotBefore attribute in decoded assertion", url, "nb|" + tokenTag);
125​
126    // 4. Flow binding.
127    if (isResponse && !/InResponseTo\s*=/.test(xml))
128        report("low", "saml-no-inresponseto",
129            "SAML Response has no InResponseTo (unsolicited-response / replay surface)",
130            "no InResponseTo attribute", url, "irt|" + tokenTag);
131    if (!/Destination\s*=/.test(xml) && isResponse)
132        report("low", "saml-no-destination",
133            "SAML Response has no Destination attribute (audience not bound to the ACS URL)",
134            "no Destination attribute", url, "dst|" + tokenTag);
135​
136    // 5. Cleartext ACS / Destination / Recipient.
137    var cleartext = xml.match(/(?:Destination|Recipient|AssertionConsumerServiceURL)\s*=\s*["'](http:\/\/[^"']+)["']/);
138    if (cleartext)
139        report("medium", "saml-cleartext-endpoint",
140            "SAML endpoint is cleartext http:// -- assertion delivered without TLS",
141            clip(cleartext[0], 120), url, "clr|" + tokenTag);
142}

144nullock.onResponse(function(entry) {
145    try {
146        var body = entry.bodyText || entry.bodyPreview || "";
147        if (!body || body.indexOf("SAML") < 0) return entry;
148        var blobs = extractSaml(body);
149        for (var i = 0; i < blobs.length; i++) {
150            var xml;
151            try { xml = b64ToStr(blobs[i].b64); } catch (e) { continue; }
152            if (xml.indexOf("<") < 0) continue;             // not XML; skip
153            var tag = (entry.url || "") + "|" + blobs[i].kind + "|" + blobs[i].b64.length;
154            auditXml(xml, blobs[i].kind, entry.url || "", tag);
155        }
156    } catch (e) {
157        nullock.log("sammy: " + (e && e.message ? e.message : String(e)));
158    }
159    return entry;
160});
