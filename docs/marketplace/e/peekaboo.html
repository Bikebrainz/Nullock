Peekaboo

Peeks into ASP.NET __VIEWSTATE: decodes it and flags the unencrypted case (ObjectStateFormatter marker + readable) -- state exposed, and a deserialization/RCE surface if EnableViewStateMac is off. Encrypted ViewState is left alone. Observe-only.

1// peekaboo.js -- passive ASP.NET __VIEWSTATE auditor.
2//
3// ASP.NET Web Forms round-trips page state through a base64 __VIEWSTATE
4// field. When it isn't encrypted (ViewStateEncryptionMode) it's readable --
5// leaking control tree, data-bound values and internal structure -- and when
6// it also isn't MAC-protected (EnableViewStateMac) it's *deserialized*
7// server-side, which is the ObjectStateFormatter / ysoserial.net RCE path.
8// This is the passive detection half of the "ViewState" BApp family, and
9// nothing in Nullock's native scanner looks at ViewState today.
10//
11// Observe-only: onResponse over response HTML, no mutation, no permission.
12//
13// What it does: find __VIEWSTATE in a served form, base64-decode it, and
14// classify:
15//   - Starts with the ObjectStateFormatter marker (0xFF 0x01) and decodes to
16//     readable content => UNENCRYPTED ViewState: contents exposed, and a
17//     tamper/deserialization surface if MAC is off. Reported.
18//   - High-entropy / no marker => encrypted; not reported (nothing to see).
19// Also notes when __VIEWSTATEGENERATOR is present (needed to weaponise an
20// unprotected ViewState with ysoserial.net --generator).
21//
22// Limits: MAC presence can't be proven from the blob alone (the HMAC is a
23// trailing opaque tail); "unencrypted + readable" is the reliable passive
24// signal, so the finding says to CONFIRM EnableViewStateMac before assuming
25// RCE. Only the first 64 KiB of the body is visible.

31function b64ToBytes(str) {
32    var clean = String(str).replace(/[^A-Za-z0-9+/]/g, ""), out = [];
33    for (var i = 0; i < clean.length; i += 4) {
34        var has2 = i + 2 < clean.length, has3 = i + 3 < clean.length;
35        var e0 = B64.indexOf(clean[i]), e1 = B64.indexOf(clean[i + 1]);
36        var e2 = has2 ? B64.indexOf(clean[i + 2]) : 0, e3 = has3 ? B64.indexOf(clean[i + 3]) : 0;
37        var n = (e0 << 18) | (e1 << 12) | ((e2 < 0 ? 0 : e2) << 6) | (e3 < 0 ? 0 : e3);
38        out.push((n >> 16) & 0xff);
39        if (has2) out.push((n >> 8) & 0xff);
40        if (has3) out.push(n & 0xff);
41    }
42    return out;
43}

60function extractField(body, field) {
61    var re = new RegExp("name\\s*=\\s*[\"']?" + field + "[\"']?[^>]*?value\\s*=\\s*[\"']([^\"']*)[\"']", "i");

69// Fraction of bytes that are printable ASCII (rough "is this readable?").
70function printableRatio(bytes, cap) {
71    var n = Math.min(bytes.length, cap || 512), p = 0;
72    if (!n) return 0;
73    for (var i = 0; i < n; i++) { var b = bytes[i]; if ((b >= 0x20 && b <= 0x7e) || b === 0x09) p++; }
74    return p / n;
