Peep

Plants OAST canaries in the headers servers tend to trust (X-Forwarded-For/Host, Referer, True-Client-IP...) so blind SSRF and out-of-band interactions ping your OAST sink. Default-OFF: a no-op until you set oastDomain. (Our Collaborator Everywhere.)

1// peep.js -- decorate outbound traffic with OAST canaries.
2//
3// nullock:permissions modify-requests
4// nullock:version 0.1.0
5//
6// A port of James Kettle's "Collaborator Everywhere" (PortSwigger's own
7// flagship BApp). The idea: a lot of blind server-side vulnerabilities --
8// SSRF, blind XXE, misrouted proxies, analytics that re-fetch a Referer,
9// password-reset links built from a Host header -- only reveal themselves
10// when the server reaches back out to a URL you planted. So plant one in
11// every request, in the headers servers are most likely to trust and act
12// on, and watch your OAST sink for the callbacks.
13//
14// Nullock ships its own OAST server + correlator (the half of Collaborator
15// that's actually worth money -- see Src/Core/APIs/Oast). This extension is
16// the *injection* half: it appends `<token>.<yourOastDomain>` into a curated
17// set of "the server tends to trust this" headers on matching requests, and
18// logs each injection so you can line callbacks up against the host you hit.
19//
20// SETUP -- required, and it is a DEFAULT-OFF no-op until you do it:
21//   1. Stand up the OAST sink (see DEPLOY_OAST.md) and note its domain.
22//   2. Put that domain in CONFIG.oastDomain below.
23//   3. Optionally scope CONFIG.onlyHosts so you don't decorate every host
24//      you browse -- injecting canaries into third-party traffic is noisy
25//      and, on targets you're not authorised to test, out of scope.
26//   4. Browse / scan as normal; poll the OAST sink (/api/oast/poll) for hits.
27//
28// Correlation note: because the sandbox can't call the C++ correlator's
29// registerToken() directly, a callback shows up as a raw OAST hit rather
30// than an auto-linked "confirmed" finding. The per-injection log line
31// (host + token + header) is what you match it against. Exposing an
32// `nullock.oastMint(context)` bridge call would make these first-class --
33// a good candidate for the JS-bridge work.
34//
35// Safety: injection only rewrites request headers (never the body or
36// method), only fires when oastDomain is set, respects onlyHosts, and skips
37// hosts that ARE your OAST domain so canaries never chase their own tail.

39(function() {
40'use strict';

42var CONFIG = {
43    // Your OAST sink's domain, e.g. "oast.mytunnel.example". BLANK => disabled.
44    oastDomain: "",
45​
46    // Only decorate requests to these hosts (exact or *.suffix). Empty =>
47    // every host except the OAST domain itself. Keep this tight in the wild.
48    onlyHosts: [],
49​
50    // Headers to plant a canary in. These are the classic "server re-uses
51    // this to make a decision or a request" set. Value-carrying headers
52    // (Referer / X-Forwarded-Host) embed a full https:// URL; IP-shaped
53    // headers get the bare hostname so they still parse as a host.
54    urlHeaders: ["Referer", "X-Forwarded-Host", "X-Host", "X-Forwarded-Server",
55                 "X-Original-URL", "X-Rewrite-URL", "Origin"],
56    hostHeaders: ["X-Forwarded-For", "True-Client-IP", "X-Client-IP",
57                  "X-Real-IP", "Forwarded", "CF-Connecting-IP"],
58​
59    // Cap injections/host so a long crawl doesn't spray thousands of tokens.
60    maxPerHost: 40,
61};
62​
63var perHostCount = {};
64​
65function hostMatches(host) {
66    if (!CONFIG.onlyHosts.length) return true;
67    for (var i = 0; i < CONFIG.onlyHosts.length; i++) {
68        var rule = CONFIG.onlyHosts[i].toLowerCase();
69        if (rule.charAt(0) === "*" && rule.charAt(1) === ".") {
70            var suf = rule.substring(1); // ".example.com"
71            if (host === rule.substring(2) || host.slice(-suf.length) === suf) return true;
72        } else if (host === rule) {
73            return true;
74        }
75    }
76    return false;
77}

79// A short, unique-ish, DNS-label-safe token. Not crypto-strong -- it only
80// needs to be collision-resistant enough to tell two callbacks apart.
81function mintToken(host) {
82    var t = "";
83    while (t.length < 12) t += Math.floor(Math.random() * 36).toString(36);
84    t = t.substring(0, 12);
85    // Fold a hint of the host in so a raw OAST hit is easier to eyeball.
86    var tag = host.replace(/[^a-z0-9]/gi, "").toLowerCase().substring(0, 8);
87    return (tag ? tag + "-" : "") + t;
88}

90function setHeader(headers, name, val) {
91    var low = name.toLowerCase();
92    for (var i = 0; i < headers.length; i++)
93        if (headers[i][0].toLowerCase() === low) { headers[i][1] = val; return; }
94    headers.push([name, val]);
95}

97nullock.onRequest(function(req) {
98    try {
99        if (!CONFIG.oastDomain) return req;                 // disabled
100        var host = (req.host || "").toLowerCase();
101        if (!host || host === CONFIG.oastDomain.toLowerCase()) return req;
102        if (host.slice(-CONFIG.oastDomain.length) === CONFIG.oastDomain.toLowerCase()) return req;
103        if (!hostMatches(host)) return req;
104        if ((perHostCount[host] || 0) >= CONFIG.maxPerHost) return req;
105​
106        var token = mintToken(host);
107        var fqdn = token + "." + CONFIG.oastDomain;
108​
109        var i;
110        for (i = 0; i < CONFIG.urlHeaders.length; i++)
111            setHeader(req.headers, CONFIG.urlHeaders[i], "https://" + fqdn + "/");
112        for (i = 0; i < CONFIG.hostHeaders.length; i++)
113            setHeader(req.headers, CONFIG.hostHeaders[i], fqdn);
114​
115        perHostCount[host] = (perHostCount[host] || 0) + 1;
116        nullock.log("peep: " + host + " <= " + fqdn
117                    + " (" + (req.method || "?") + " " + (req.path || "/") + ")");
118        return req;
119    } catch (e) {
120        nullock.log("peep: " + (e && e.message ? e.message : String(e)));
121        return req;
122    }
123});

133    : "peep: loaded but DISABLED (set CONFIG.oastDomain to arm)");

135})();
