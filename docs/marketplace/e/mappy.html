Mappy

Flags exposed JS/CSS source maps -- external .map references, inline data: maps, or a .map served with sourcesContent (original-source leak). Where every 'unpack the bundle' recon starts. Observe-only.

1// mappy.js -- flag exposed JavaScript/CSS source maps.
2//
3// Source maps are a build-time convenience that routinely ship to prod by
4// accident. A reachable .map file hands an attacker your ORIGINAL source --
5// pre-minification names, code comments, dead/feature-flagged branches,
6// internal file paths, sometimes hard-coded endpoints or secrets. Tools
7// that "unpack webpack" (and half of every recon methodology) start here.
8// Nullock's native scanner reads served JS for secrets and links but says
9// nothing about source-map exposure, so this fills that gap.
10//
11// Observe-only: onResponse, no mutation, no permission.
12//
13// What it reports:
14//   - A served asset referencing an external .map via
15//     `//# sourceMappingURL=...` (original source likely one fetch away).
16//   - An inline `data:` source map embedded in the asset (source is right
17//     there in the response -- no second request needed).
18//   - A .map served directly with `sourcesContent` (CONFIRMED source leak:
19//     the original files are inlined in the map).
20//
21// Limits: for the external-reference case it flags the *reference*; it can't
22// fetch the .map to confirm it 200s (passive, single-message). The direct-
23// .map and inline-data cases are confirmed from the body we already have.

26'use strict';

28function headerVal(headers, name) {
29    if (!headers) return "";
30    var low = name.toLowerCase();
31    for (var i = 0; i < headers.length; i++)
32        if (headers[i][0] && headers[i][0].toLowerCase() === low) return headers[i][1] || "";
33    return "";
34}
35function clip(s, n) { s = String(s || ""); return s.length > n ? s.substring(0, n) + "..." : s; }

37var seen = {};
38function report(sev, kind, summary, evidence, url, key) {
39    if (seen[key]) return; seen[key] = 1;
40    try { nullock.reportFinding(sev, kind, summary, evidence, url); }
41    catch (e) { nullock.log("mappy report failed: " + e); }
42}

44nullock.onResponse(function(entry) {
45    try {
46        var url = entry.url || "";
47        var ct = headerVal(entry.headers, "Content-Type").toLowerCase();
48        var body = entry.bodyText || entry.bodyPreview || "";
49        if (!body) return entry;
50​
51        var isJs  = ct.indexOf("javascript") >= 0 || ct.indexOf("ecmascript") >= 0 || /\.m?js(\?|$)/i.test(url);
52        var isCss = ct.indexOf("css") >= 0 || /\.css(\?|$)/i.test(url);
53        var isMap = ct.indexOf("json") >= 0 || /\.map(\?|$)/i.test(url) || /^\s*\{\s*"version"\s*:\s*3/.test(body);
54​
55        // 1. A .map served directly and carrying inlined original sources.
56        if (isMap && /"sourcesContent"\s*:/.test(body) && /"version"\s*:\s*3/.test(body)) {
57            var srcCount = (body.match(/"sources"\s*:\s*\[([^\]]*)\]/) || [,""])[1].split(",").length;
58            report("medium", "sourcemap-exposed",
59                "Source map served with inlined original sources (sourcesContent) -- original code is exposed",
60                "~" + srcCount + " source file(s) inlined in the served .map", url, "map|" + url);
61            return entry;
62        }
63​
64        if (!isJs && !isCss) return entry;
65​
66        // 2. sourceMappingURL directive in a served asset.
67        var m = body.match(/[#@]\s*sourceMappingURL\s*=\s*([^\s'"]+)/);
68        if (m) {
69            var ref = m[1];
70            if (/^data:/i.test(ref)) {
71                report("medium", "sourcemap-inline-data",
72                    "Asset embeds an inline data: source map -- original source is inside the response",
73                    "sourceMappingURL=data:... (" + ref.length + " bytes)", url, "inline|" + url);
74            } else {
75                report("low", "sourcemap-reference-exposed",
76                    "Asset references an external source map (" + clip(ref, 60) + ") -- fetch it to confirm source exposure",
77                    "sourceMappingURL=" + clip(ref, 100), url, "ref|" + url + "|" + ref);
78            }
79        }
80    } catch (e) {
81        nullock.log("mappy: " + (e && e.message ? e.message : String(e)));
82    }
83    return entry;
84});
