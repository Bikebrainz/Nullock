DOM Taint Tracker

Static DOM-XSS taint analysis on served HTML/JS with one-hop dataflow: connects sources (location.*, referrer, window.name, postMessage, storage) to sinks (innerHTML, eval, document.write, setTimeout, location-nav) and emits findings. The proxy-side answer to Burp DOM Invader.

1// dom_taint.js -- static DOM-XSS taint analysis on served HTML/JS.
2//
3// nullock:version 0.2.0
4//
5// Burp Pro's DOM Invader runs *inside* the browser and instruments the
6// live DOM at runtime. Nullock is a proxy, so we can't hook the page --
7// but every script the app serves passes through us, and we can analyse
8// it statically. This extension does source -> sink taint matching on
9// each HTML/JS response and emits findings when attacker-controllable
10// input can reach a dangerous sink.
11//
12// Beyond naive co-occurrence, we do ONE-HOP dataflow: if a source is
13// assigned into a variable (`var h = location.hash`) and that variable
14// then reaches a sink (`el.innerHTML = h`), we connect them and report
15// the *original* source -- not whatever token happened to sit nearest
16// the sink. Two propagation passes catch `y = x` chains as well.
17//
18// What this catches:
19//   - innerHTML / outerHTML / document.write / insertAdjacentHTML fed
20//     (directly or via a 1-2 hop variable) from location.* /
21//     document.referrer / window.name / postMessage / web-storage
22//   - eval / Function / setTimeout(string) fed from the same
23//   - location assignment / element.src (DOM open-redirect) from a source
24//
25// Honest limits of static analysis:
26//   - Taint through 3+ variable hops, function params, or object props
27//     is not followed (we do shallow, single-statement dataflow)
28//   - Sinks reached only via dynamically built code aren't seen
29//   - Anything past the 64 KiB body preview the proxy hands us
30// Treat findings as leads to confirm in a browser, like Burp's DOM hints.

35// ---- source / sink vocabularies ---------------------------------------
36// Sources: values an attacker can influence without touching the server.
37// `re` is used global (offset scan) and reset before each .test().
38var SOURCES = [
39    { re: /location\s*\.\s*(?:href|search|hash|pathname|host|hostname)/g, name: "location.*" },
40    { re: /document\s*\.\s*(?:URL|documentURI|baseURI)/g,                name: "document.URL" },
41    { re: /document\s*\.\s*referrer/g,                                   name: "document.referrer" },
42    { re: /\bwindow\s*\.\s*name\b/g,                                     name: "window.name" },
43    { re: /document\s*\.\s*cookie/g,                                     name: "document.cookie" },
44    { re: /\.(?:getItem)\s*\(/g,                                         name: "web-storage" },
45    { re: /\baddEventListener\s*\(\s*['"]message['"]/g,                  name: "postMessage" },
46    { re: /\bonmessage\b/g,                                              name: "postMessage" },
47    // Bare `location` last so the qualified forms above win attribution.
48    { re: /\blocation\b(?!\s*\.\s*(?:protocol|origin)\b)/g,              name: "location" },
49];

51// Sinks: APIs that execute code or inject markup. The capture after the
52// sink token (RHS of `=`, or call arguments) is the tainted region.
53var SINKS = [
54    { re: /\.\s*innerHTML\s*=/g,                 name: "innerHTML",          cls: "html" },
55    { re: /\.\s*outerHTML\s*=/g,                 name: "outerHTML",          cls: "html" },
56    { re: /\.\s*insertAdjacentHTML\s*\(/g,       name: "insertAdjacentHTML", cls: "html" },
57    { re: /document\s*\.\s*write(?:ln)?\s*\(/g,  name: "document.write",     cls: "html" },
58    { re: /\.\s*html\s*\(/g,                      name: "jQuery.html",        cls: "html" },
59    { re: /\bcreateContextualFragment\s*\(/g,    name: "createContextualFragment", cls: "html" },
60    { re: /\beval\s*\(/g,                         name: "eval",               cls: "code" },
61    { re: /\bnew\s+Function\s*\(/g,              name: "Function",           cls: "code" },
62    { re: /\bsetTimeout\s*\(/g,                   name: "setTimeout",         cls: "code" },
63    { re: /\bsetInterval\s*\(/g,                  name: "setInterval",        cls: "code" },
64    { re: /location\s*\.\s*(?:href|assign|replace)\s*[=(]/g, name: "location-nav", cls: "nav" },
65    { re: /\.\s*src\s*=/g,                         name: "element.src",        cls: "nav" },

74// Bounded de-dup so a reloaded page doesn't re-report the same flow.
75var seen = {};
76var SEEN_CAP = 5000;
77function alreadyReported(key) {
78    if (seen[key]) return true;
79    seen[key] = 1;
80    var n = 0; for (var k in seen) { if (++n > SEEN_CAP) { seen = {}; break; } }
81    return false;
82}

84// ---- helpers -----------------------------------------------------------
85function headerVal(headers, name) {
86    if (!headers) return "";
87    var low = name.toLowerCase();
88    for (var i = 0; i < headers.length; i++) {
89        if (headers[i][0] && headers[i][0].toLowerCase() === low)
90            return headers[i][1] || "";
91    }
92    return "";
93}

95// Returns the source name if `text` contains any source token, else "".
96function sourceIn(text) {
97    for (var s = 0; s < SOURCES.length; s++) {
98        SOURCES[s].re.lastIndex = 0;
99        if (SOURCES[s].re.test(text)) return SOURCES[s].name;
100    }
101    return "";
102}

104// Returns the origin-source name if `text` references any tainted var.
105function taintedVarIn(text, taint) {
106    for (var v in taint) {
107        // word-boundary match so `h` doesn't match `height`
108        var re = new RegExp("(?:^|[^\\w$])" + v.replace(/[.*+?^${}()|[\]\\]/g, "\\$&") + "(?:[^\\w$]|$)");
109        if (re.test(text)) return taint[v];
110    }
111    return "";
112}

114// The tainted region for a sink at `idx`: everything from the sink token
115// to the end of its statement (next ; or newline or block close), capped.
116function sinkRegion(code, afterIdx) {
117    var end = code.length;
118    for (var i = afterIdx; i < code.length && i < afterIdx + 300; i++) {
119        var ch = code[i];
120        if (ch === ';' || ch === '\n' || ch === '}') { end = i; break; }
121    }
122    return code.substring(afterIdx, Math.min(end, afterIdx + 300));
123}

125// Collect variables assigned (directly or transitively) from a source.
126// taint[varName] = originSourceName. Two passes to catch `y = x` chains.
127function collectTaint(code) {
128    var taint = {};
129    var assignRe = /(?:\bvar\b|\blet\b|\bconst\b)?\s*([A-Za-z_$][\w$]*)\s*=\s*([^;\n]{0,240})/g;
130    for (var pass = 0; pass < 2; pass++) {
131        assignRe.lastIndex = 0;
132        var m;
133        while ((m = assignRe.exec(code)) !== null) {
134            var lhs = m[1], rhs = m[2];
135            if (taint[lhs]) continue;
136            var src = sourceIn(rhs);
137            if (src) { taint[lhs] = src; continue; }
138            if (pass > 0) {
139                var via = taintedVarIn(rhs, taint);
140                if (via) taint[lhs] = via;
141            }
142        }
143    }
144    return taint;
145}

147function offsets(re, code) {
148    re.lastIndex = 0;
149    var out = [], m;
150    while ((m = re.exec(code)) !== null) {
151        out.push(re.lastIndex);   // position AFTER the sink token
152        if (re.lastIndex === m.index) re.lastIndex++;
153    }
154    return out;
155}

157function snippet(code, idx) {
158    var a = Math.max(0, idx - 50), b = Math.min(code.length, idx + 90);
159    return code.substring(a, b).replace(/\s+/g, " ").substring(0, 200);
160}

162// ---- script context extraction ----------------------------------------
163function scriptContexts(body, contentType, url) {
164    var ct = (contentType || "").toLowerCase();
165    var isJs = ct.indexOf("javascript") >= 0 || ct.indexOf("ecmascript") >= 0
166            || /\.m?js(\?|$)/i.test(url);
167    if (isJs) return [{ code: body, where: "script body" }];
168​
169    var isHtml = ct.indexOf("html") >= 0 || /<html|<!doctype/i.test(body);
170    if (!isHtml) return [];
171​
172    var ctxs = [];
173    var re = /<script\b([^>]*)>([\s\S]*?)<\/script>/gi, m;
174    while ((m = re.exec(body)) !== null) {
175        if (/\bsrc\s*=/i.test(m[1])) continue;   // external; analysed when fetched
176        if (m[2] && m[2].trim().length) ctxs.push({ code: m[2], where: "inline <script>" });
177    }
178    var reAttr = /\bon[a-z]+\s*=\s*("(?:[^"]*)"|'(?:[^']*)')/gi;
179    while ((m = reAttr.exec(body)) !== null) {
180        ctxs.push({ code: m[1], where: "inline event-handler" });
181    }
182    return ctxs;
183}

185// ---- main pass ---------------------------------------------------------
186function analyse(entry) {
187    var body = entry.bodyText || entry.bodyPreview || "";
188    if (!body) return;
189    var url = entry.url || "";
190    var ct = headerVal(entry.headers, "Content-Type");
191​
192    var contexts = scriptContexts(body, ct, url);
193    for (var c = 0; c < contexts.length; c++) {
194        var code = contexts[c].code, where = contexts[c].where;
195        var taint = collectTaint(code);
196​
197        for (var k = 0; k < SINKS.length; k++) {
198            var sink = SINKS[k];
199            var ends = offsets(sink.re, code);
200            for (var j = 0; j < ends.length; j++) {
201                var region = sinkRegion(code, ends[j]);
202​
203                // Direct source in the sink's own expression -> high
204                // confidence. Else a tainted variable reaching it -> still
205                // a real flow, attribute to the variable's origin source.
206                var src = sourceIn(region);
207                var direct = !!src;
208                if (!src) src = taintedVarIn(region, taint);
209                if (!src) continue;   // no taint reaches this sink; skip
210​
211                var meta = CLASS_META[sink.cls];
212                var sev = (sink.cls === "nav" && !direct) ? "medium"
213                        : (sink.cls === "nav") ? "medium" : "high";
214​
215                var key = url + "|" + sink.name + "|" + src + "|" + where;
216                if (alreadyReported(key)) continue;
217​
218                var summary = meta.label + ": " + src + " -> " + sink.name
219                            + (direct ? "" : " (via variable)");
220                var evidence = "[" + where + "] " + snippet(code, ends[j]);
221                try {
222                    nullock.reportFinding(sev, "dom-taint-" + sink.cls,
223                                          summary, evidence, url);
224                } catch (e) {
225                    nullock.log("dom_taint report failed: " + e);
226                }
227            }
228        }
229    }
230}

232nullock.onResponse(function(entry) {
233    try { analyse(entry); }
234    catch (e) {
235        nullock.log("dom_taint: " + (e && e.message ? e.message : String(e)));
236    }
237    return entry;   // read-only; we never mutate
238});
