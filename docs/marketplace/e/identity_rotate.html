1// identity_rotate.js -- per-request identity rotation.

3// nullock:permissions modify-requests
4//
5// Rotates User-Agent, Accept-Language, Sec-CH-UA, and a handful of
6// other browser-fingerprint headers across a pool of realistic
7// browsers. Useful for testing apps that key on user identity for
8// rate-limiting, A/B-bucket assignment, or bot detection.
9//
10// Caveat: this is HTTP-level rotation only. JA3 / JA4 (TLS handshake
11// fingerprint) is unchanged; Qt's QSslSocket doesn't expose enough
12// to reorder TLS extensions and SChannel ignores the order anyway.
13// A WAF that fingerprints TLS will still see a "Qt-shaped" handshake.
14// That's a fundamental architectural limit -- a real fix would
15// require swapping the outbound TLS stack to something like utls.

61// Sticky per-host: requests to the same host get the same identity until
62// the user explicitly rotates. Otherwise a single browsing session looks
63// schizophrenic to the target.
64var STICKY = {};

66function pickFor(host) {
67    if (STICKY[host]) return STICKY[host];
68    var p = PROFILES[Math.floor(Math.random() * PROFILES.length)];
69    STICKY[host] = p;
70    return p;
71}

73function setHeader(headers, name, val) {
74    var nameLow = name.toLowerCase();
75    for (var i = 0; i < headers.length; i++) {
76        if (headers[i][0].toLowerCase() === nameLow) {
77            headers[i][1] = val;
78            return;
79        }
80    }
81    headers.push([name, val]);
82}
83​
84function dropHeader(headers, name) {
85    var nameLow = name.toLowerCase();
86    for (var i = headers.length - 1; i >= 0; i--) {
87        if (headers[i][0].toLowerCase() === nameLow) headers.splice(i, 1);
88    }
89}

91nullock.onRequest(function(req) {
92    try {
93        var prof = pickFor(req.host);
94        setHeader(req.headers, "User-Agent", prof.ua);
95        setHeader(req.headers, "Accept", prof.accept);
96        setHeader(req.headers, "Accept-Language", prof.acceptLang);
97        // Chromium-only client hints. For Firefox/Safari we drop them
98        // entirely so the request looks consistent.
99        if (prof.secChUa) {
100            setHeader(req.headers, "Sec-CH-UA", prof.secChUa);
101            setHeader(req.headers, "Sec-CH-UA-Mobile", prof.secChUaMobile);
102            setHeader(req.headers, "Sec-CH-UA-Platform", prof.secChUaPlatform);
103        } else {
104            dropHeader(req.headers, "Sec-CH-UA");
105            dropHeader(req.headers, "Sec-CH-UA-Mobile");
106            dropHeader(req.headers, "Sec-CH-UA-Platform");
107        }
108        return req;
109    } catch (e) {
110        nullock.log("identity_rotate: " + (e && e.message ? e.message : String(e)));
111        return req;
112    }
113});
