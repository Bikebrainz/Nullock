Crumbs

Picks endpoint & route crumbs out of served JavaScript -- paths, fetch()/axios() targets, route strings -- and surfaces the sensitive-looking ones (/admin, /internal, /debug, versioned APIs...) on their own. The recon map nothing links to. (LinkFinder/GAP-style.) Observe-only.

1// crumbs.js -- endpoint & route crumbs picked out of served JavaScript.
2//
3// nullock:version 0.1.0
4//
5// The recon staple: modern apps ship their whole API surface in the JS
6// bundle. Pull the paths, fetch()/axios() targets and route strings out of
7// every script that flows past, and you get a map of endpoints to go poke --
8// including the /admin, /internal and /debug ones nothing links to. This is
9// the job LinkFinder / GAP do; Nullock's native JS pass hunts secrets, not
10// routes, so this is a clean addition rather than a duplicate.
11//
12// Observe-only: onResponse over JS (and inline <script>), no permission.
13//
14// It reports two ways so it stays high-signal:
15//   - One `js-endpoints-discovered` info finding per script summarising the
16//     count with a sample (the full list goes to the extension log).
17//   - A `js-endpoint-sensitive` low finding for any path that smells juicy
18//     (admin / internal / debug / token / upload / actuator / graphql /
19//     swagger / api-key / a versioned /v1 API ...), so those surface on their
20//     own instead of drowning in the summary.
21//
22// Limits: static string-scraping. It won't see endpoints built by
23// concatenation at runtime, and it will occasionally grab a non-endpoint
24// string that happens to look like a path. Treat it as leads to explore.

49// Endpoint extraction patterns over a chunk of JS/HTML.
50function extract(code) {
51    var found = {};
52    function add(p) {
53        if (!p) return;
54        p = p.trim();
55        if (p.length < 2 || p.length > 180) return;
56        if (BORING.test(p)) return;
57        if (/^https?:\/\//i.test(p) || /^\//.test(p)) found[p] = 1;
58    }
59    var m, re;
60    // Quoted absolute-path or URL literals: "/api/v2/users", 'https://x/y'
61    re = /["'`](https?:\/\/[^"'`\s]{4,180}|\/[A-Za-z0-9_~%\-./]{2,180})["'`]/g;
62    while ((m = re.exec(code)) !== null) add(m[1]);
63    // fetch("...") / axios.get("...") / .open(method, "...") targets
64    re = /(?:fetch|axios(?:\.\w+)?|\.open|url\s*[:=]|href\s*[:=]|api\s*[:=])\s*\(?\s*["'`]([^"'`]{2,180})["'`]/g;
65    while ((m = re.exec(code)) !== null) add(m[1]);
66    var out = Object.keys(found);
67    // Drop obvious non-endpoints: pure "//", protocol-relative junk, regexes.
68    out = out.filter(function(p){ return !/^\/\//.test(p) && !/[<>{}]/.test(p); });
69    return out;
70}

72function scriptChunks(body, ct, url) {
73    var isJs = ct.indexOf("javascript") >= 0 || ct.indexOf("ecmascript") >= 0 || /\.m?js(\?|$)/i.test(url);
74    if (isJs) return [body];
75    if (ct.indexOf("html") >= 0 || /<html|<!doctype/i.test(body)) {
76        var chunks = [], re = /<script\b[^>]*>([\s\S]*?)<\/script>/gi, m;
77        while ((m = re.exec(body)) !== null) if (m[1] && m[1].trim()) chunks.push(m[1]);
78        return chunks;
79    }
80    return [];
81}

98        // Sensitive ones get their own low finding.
99        var juicy = list.filter(function(p){ return SENSITIVE.test(p); });
100        for (var k = 0; k < juicy.length && k < 25; k++) {
101            report("low", "js-endpoint-sensitive",
102                "Sensitive-looking endpoint referenced in JS: " + clip(juicy[k], 100),
103                "source: " + clip(url, 90), url, "juicy|" + url + "|" + juicy[k]);
104        }
105        // One rolled-up info finding for the rest.
106        var sample = list.slice(0, 12).map(function(p){ return clip(p, 80); }).join("  ");
107        report("info", "js-endpoints-discovered",
108            list.length + " endpoint(s) extracted from JavaScript"
109            + (juicy.length ? " (" + juicy.length + " sensitive-looking)" : ""),
110            "sample: " + sample + (list.length > 12 ? "  (+" + (list.length - 12) + " more in ext log)" : ""),
111            url, "sum|" + url + "|" + list.length);
112        if (list.length > 12) nullock.log("crumbs: " + url + " -> " + list.join("  "));
113    } catch (e) {
114        nullock.log("crumbs: " + (e && e.message ? e.message : String(e)));
115    }
116    return entry;
